cmake_minimum_required(VERSION 3.20)
project(ShoenDomainCore LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(DOMAIN_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../game/Source/DomainCore")
add_library(domain_core STATIC "${DOMAIN_SOURCE}/Private/sim/World.cpp" "${DOMAIN_SOURCE}/Private/sim/SaveCodec.cpp" "${DOMAIN_SOURCE}/Private/sim/Battle.cpp")
target_include_directories(domain_core PUBLIC "${DOMAIN_SOURCE}/Public")
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(domain_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(domain_core_tests tests/test_core.cpp)
target_link_libraries(domain_core_tests PRIVATE domain_core)
enable_testing()
add_test(NAME domain_core_invariants COMMAND domain_core_tests)
