cmake_minimum_required(VERSION 3.20)
project(ShoenDomainCore LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(DOMAIN_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../game/Source/DomainCore")
add_library(domain_core STATIC "${DOMAIN_SOURCE}/Private/sim/World.cpp" "${DOMAIN_SOURCE}/Private/sim/SaveCodec.cpp" "${DOMAIN_SOURCE}/Private/sim/Battle.cpp" "${DOMAIN_SOURCE}/Private/sim/Buildings.cpp" "${DOMAIN_SOURCE}/Private/sim/Inspection.cpp" "${DOMAIN_SOURCE}/Private/sim/ProfilingFixture.cpp")
target_include_directories(domain_core PUBLIC "${DOMAIN_SOURCE}/Public")
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(domain_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
target_sources(domain_core PRIVATE "${DOMAIN_SOURCE}/Private/sim/Prototype.cpp")
add_executable(domain_core_tests tests/test_core.cpp)
target_link_libraries(domain_core_tests PRIVATE domain_core)
enable_testing()
add_test(NAME domain_core_invariants COMMAND domain_core_tests)

add_executable(domain_building_tests tests/test_buildings.cpp)
target_link_libraries(domain_building_tests PRIVATE domain_core)
target_compile_definitions(domain_building_tests PRIVATE FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
add_test(NAME domain_building_invariants COMMAND domain_building_tests)

add_executable(domain_inspection_tests tests/test_inspection.cpp)
target_link_libraries(domain_inspection_tests PRIVATE domain_core)
add_test(NAME domain_inspection_invariants COMMAND domain_inspection_tests)

add_executable(domain_profiling_fixture_tests tests/test_profiling_fixture.cpp)
target_link_libraries(domain_profiling_fixture_tests PRIVATE domain_core)
add_test(NAME domain_profiling_fixture_invariants COMMAND domain_profiling_fixture_tests)

add_executable(domain_prototype_tests tests/test_prototype.cpp)
target_link_libraries(domain_prototype_tests PRIVATE domain_core)
add_test(NAME domain_prototype_invariants COMMAND domain_prototype_tests)

target_sources(domain_core PRIVATE "${DOMAIN_SOURCE}/Private/sim/Terrain.cpp")
add_executable(domain_terrain_tests tests/test_terrain.cpp)
target_link_libraries(domain_terrain_tests PRIVATE domain_core)
add_test(NAME domain_terrain_invariants COMMAND domain_terrain_tests)

add_executable(domain_terrain_suitability_tests tests/test_terrain_suitability.cpp)
target_link_libraries(domain_terrain_suitability_tests PRIVATE domain_core)
add_test(NAME domain_terrain_suitability_invariants COMMAND domain_terrain_suitability_tests)
